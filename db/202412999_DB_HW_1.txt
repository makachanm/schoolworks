create table movies (movie_id int primary key, title varchar(100), genre varchar(30), director varchar(50), release_year int, ticket_price int);
create table users(viewer_id varchar(20) primary key, viewer_name varchar(50), city varchar(30), grade varchar(20));
create table reservation(reserve_id int primary key, viewer_id varchar(20), movie_id int, reserve_date date, pos int, foreign key(viewer_id) references users(viewer_id), foreign key(movie_id) references movies(movie_id));
select genre, count(genre) as counts from movies group by genre order by counts desc;
